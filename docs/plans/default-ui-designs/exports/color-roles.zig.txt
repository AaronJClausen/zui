pub const ColorRole = enum {
    background,
    surface,
    surface_raised,
    surface_sunken,
    surface_overlay,
    surface_inverse,
    on_surface,
    on_surface_muted,
    on_surface_inverse,
    on_surface_accent,
    accent,
    on_accent,
    accent_container,
    on_accent_container,
    secondary_container,
    on_secondary_container,
    danger,
    on_danger,
    danger_container,
    on_danger_container,
    warning,
    on_warning,
    warning_container,
    on_warning_container,
    success,
    on_success,
    success_container,
    on_success_container,
    selection,
    on_selection,
    outline,
    outline_muted,
    focus,
    scrim,
    shadow,
    rim,
};

pub fn partnerOf(role: ColorRole) ColorRole {
    return switch (role) {
        .background => .on_surface,
        .surface => .on_surface,
        .surface_raised => .on_surface,
        .surface_sunken => .on_surface,
        .surface_overlay => .on_surface,
        .surface_inverse => .on_surface_inverse,
        .on_surface => .surface,
        .on_surface_muted => .surface,
        .on_surface_inverse => .surface_inverse,
        .on_surface_accent => .surface,
        .accent => .on_accent,
        .on_accent => .accent,
        .accent_container => .on_accent_container,
        .on_accent_container => .accent_container,
        .secondary_container => .on_secondary_container,
        .on_secondary_container => .secondary_container,
        .danger => .on_danger,
        .on_danger => .danger,
        .danger_container => .on_danger_container,
        .on_danger_container => .danger_container,
        .warning => .on_warning,
        .on_warning => .warning,
        .warning_container => .on_warning_container,
        .on_warning_container => .warning_container,
        .success => .on_success,
        .on_success => .success,
        .success_container => .on_success_container,
        .on_success_container => .success_container,
        .selection => .on_selection,
        .on_selection => .selection,
        .outline => .surface,
        .outline_muted => .surface,
        .focus => .surface,
        .scrim => .surface,
        .shadow => .surface,
        .rim => .surface,
    };
}

fn kindOf(role: ColorRole) RoleKind {
    return switch (role) {
        .background, .surface, .surface_raised, .surface_sunken, .surface_overlay, .surface_inverse => .base_surface,
        .accent, .accent_container, .secondary_container, .danger, .danger_container, .warning, .warning_container, .success, .success_container, .selection => .fill,
        .scrim, .shadow, .rim => .scrim,
        else => .foreground,
    };
}

pub fn tierOf(role: ColorRole) ?Tier {
    return switch (role) {
        .on_surface, .on_surface_muted, .on_surface_inverse, .on_surface_accent, .on_accent, .on_accent_container, .on_secondary_container, .on_danger, .on_danger_container, .on_warning, .on_warning_container, .on_success, .on_success_container, .on_selection => .text,
        .outline, .focus => .non_text,
        .background, .surface, .surface_raised, .surface_sunken, .surface_overlay, .surface_inverse, .accent, .accent_container, .secondary_container, .danger, .danger_container, .warning, .warning_container, .success, .success_container, .selection, .outline_muted, .scrim, .shadow, .rim => null,
    };
}
